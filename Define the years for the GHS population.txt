// Define the years for the GHS population data
var years = ee.List.sequence(1975, 2030, 5);

// Load the FAO GAUL dataset (Admin Level 1)
var gaul_adm1 = ee.FeatureCollection("FAO/GAUL/2015/level1");

// Define the four representative cities with their respective ADM1 regions
var cities = [
  {name: "Lagos", country: "Nigeria", adm1_name: "Lagos"},
  {name: "Nairobi", country: "Kenya", adm1_name: "Nairobi"},
  {name: "Johannesburg", country: "South Africa", adm1_name: "Gauteng"},
  {name: "Kinshasa", country: "Democratic Republic of the Congo", adm1_name: "Kinshasa"}
];

// Filter GAUL ADM1 dataset to get city-level boundaries
var cityFeatures = cities.map(function(city) {
  var adm1Boundary = gaul_adm1
    .filter(ee.Filter.eq("ADM0_NAME", city.country))
    .filter(ee.Filter.eq("ADM1_NAME", city.adm1_name))
    .geometry();
  
  return ee.Feature(adm1Boundary, {city: city.name, country: city.country, adm1: city.adm1_name});
});

// Convert to FeatureCollection
var cityCollection = ee.FeatureCollection(cityFeatures);

// Function to calculate population for each ADM1 region
function calculatePopulation(feature) {
  var yearValues = {};

  // Loop through years and compute population
  years.getInfo().forEach(function (year) {
    var imagePath = "projects/sat-io/open-datasets/GHS/GHS_POP/GHS_POP_E" + year + "_GLOBE_R2023A_54009_100_V1_0";
    var popImage = ee.Image(imagePath);
    var popStat = popImage.reduceRegion({
      reducer: ee.Reducer.sum(),
      geometry: feature.geometry(),
      scale: 100,
      maxPixels: 1e13
    }).get('b1'); // Band name is 'b1'
    
    yearValues[year] = popStat; // Store value for the year
  });

  // Add each year as a separate property
  Object.keys(yearValues).forEach(function(year) {
    feature = feature.set("pop_" + year, yearValues[year]);
  });

  return feature;
}

// Compute population for each ADM1 region
var citiesWithPop = cityCollection.map(calculatePopulation);

// Print the results
print("Population statistics for selected ADM1 regions:", citiesWithPop);

var visParams = {

  max: 30,  // Adjust range based on population scale
  palette: ['blue', 'cyan', 'yellow', 'red']
};


// Loop through years to create a map layer for each population dataset
years.getInfo().forEach(function (year) {
  var popImage = ee.Image("projects/sat-io/open-datasets/GHS/GHS_POP/GHS_POP_E" + year + "_GLOBE_R2023A_54009_100_V1_0");
  
  Map.addLayer(popImage.clip(cityCollection), visParams, "Population " + year);

// Center the map on an average location
Map.centerObject(cityCollection, 4);
Map.addLayer(cityCollection.style({color: 'black', width: 2}), {}, "City Boundaries");

// === EXPORT DATA AS TIF FILES (PER CITY, PER YEAR) ===
years.getInfo().forEach(function (year) {
  cityCollection.evaluate(function(cityList) {
    cityList.features.forEach(function(cityFeature) {
      var city = ee.Feature(cityFeature);

      // ✅ Extract and format city name correctly
      var cityName = city.get("adm1").getInfo();  // Using "adm1" since that field exists
      cityName = cityName ? cityName.replace(/\s+/g, "_") : "Unknown_City";

      // ✅ Construct image path
      var popImagePath = "projects/sat-io/open-datasets/GHS/GHS_POP/GHS_POP_E" + year + "_GLOBE_R2023A_54009_100_V1_0";
      var popImage = ee.Image(popImagePath);

      // ✅ Export per city per year with proper naming
      Export.image.toDrive({
        image: popImage.clip(city.geometry()),
        description: "Population_" + year + "_" + cityName, // Now includes year & city name
        folder: "modelling",
        scale: 30,
        region: city.geometry(),
        fileFormat: "GeoTIFF",
        maxPixels: 1e13
      });
    });
  });
});


// === EXPORT DATA AS CSV ===
Export.table.toDrive({
  collection: citiesWithPop,
  description: "Population_Statistics_Cities",
  folder: "modelling",
